BASE ARCHITECTURE DE AGENTE COGNITIVO:

from abc import ABC, abstractmethod
import asyncio

class CognitiveAgent(ABC):
    def __init__(self, name: str, specialization: str):
        self.name = name
        self.specialization = specialization
        self.memory = {}
        self.action_history = []

    @abstractmethod
    async def perceive(self, input_data: Dict) -> Dict:
        pass

    @abstractmethod
    async def think(self, perception: Dict) -> Dict:
        pass

    @abstractmethod
    async def act(self, decision: Dict) -> Dict:
        pass

    async def cognitive_cycle(self, input_data: Dict) -> Dict:
        perception = await self.perceive(input_data)
        decision = await self.think(perception)
        action = await self.act(decision)

        self.action_history.append({
            'input': input_data,
            'perception': perception,
            'decision': decision,
            'action': action
        })

        return action

// Event-driven communication
interface DomainEvent {
  aggregateId: string;
  eventType: string;
  timestamp: Date;
  data: any;
}

class EventBus {
  private subscribers = new Map<string, Function[]>();

  subscribe(eventType: string, handler: Function) {
    if (!this.subscribers.has(eventType)) {
      this.subscribers.set(eventType, []);
    }
    this.subscribers.get(eventType)!.push(handler);
  }

  async publish(event: DomainEvent) {
    const handlers = this.subscribers.get(event.eventType) || [];
    for (const handler of handlers) {
      await handler(event);
    }
  }
}

import { useState, useEffect, useCallback, useRef } from 'react';

function useFetchWithCache(url: string) {
  const [data, setData] = useState(null);
  const [loading, setLoading] = useState(false);
  const cacheRef = useRef(new Map());

  const fetch = useCallback(async () => {
    if (cacheRef.current.has(url)) {
      setData(cacheRef.current.get(url));
      return;
    }

    setLoading(true);
    const response = await fetch(url);
    const json = await response.json();
    cacheRef.current.set(url, json);
    setData(json);
    setLoading(false);
  }, [url]);

  useEffect(() => { fetch(); }, [fetch]);
  return { data, loading };
}

ORQUESTRADOR MULTI-AGENTE:

class AgentOrchestrator:
    def __init__(self):
        self.agents = {}
        self.coordination_memory = {}

    def register_agent(self, agent, role: str):
        self.agents[f"{role}_{agent.config.name}"] = agent

    async def orchestrate(self, task: Dict) -> Dict:
        results = {}

        analyzer = self.agents.get('analyzer_Analyzer')
        if analyzer:
            results['analysis'] = await analyzer.process_input(task['description'])

        processor = self.agents.get('processor_Processor')
        if processor:
            results['processing'] = await processor.process_input(
                results['analysis']['response']
            )

        validator = self.agents.get('validator_Validator')
        if validator:
            results['validation'] = await validator.process_input(
                results['processing']['response']
            )

        self._share_knowledge(results)
        return results

    def _share_knowledge(self, results: Dict):
        for agent_key, agent in self.agents.items():
            for phase, result in results.items():
                agent.memory.store({'phase': phase, 'data': result, 'shared': True})

from datetime import datetime, timedelta
import hashlib

class ContextManager:
    def __init__(self, max_context_size_mb: int = 100):
        self.current_context = {}
        self.context_history = []

    def set_context(self, key: str, value, ttl_seconds=None):
        self.current_context[key] = {
            'value': value,
            'timestamp': datetime.now(),
            'ttl': ttl_seconds
        }
        self._cleanup_expired()

    def get_context(self, key: str):
        if key not in self.current_context:
            return None
        ctx = self.current_context[key]
        if ctx['ttl'] and (datetime.now() - ctx['timestamp']).total_seconds() > ctx['ttl']:
            del self.current_context[key]
            return None
        return ctx['value']

    def create_checkpoint(self) -> str:
        checkpoint = {
            'timestamp': datetime.now(),
            'context': self.current_context.copy(),
            'hash': self._calculate_hash()
        }
        self.context_history.append(checkpoint)
        return checkpoint['hash']

# BLOCO 20 - MULTI-STACK INTEGRATION PATTERNS

STACK INTEGRATOR:

import asyncio
import httpx

class StackIntegrator:
    def __init__(self):
        self.service_registry = {}

    def register_service(self, name: str, url: str):
        self.service_registry[name] = url

    async def call_service(self, service: str, endpoint: str, data: Dict) -> Dict:
        url = self.service_registry.get(service)
        if not url:
            raise ValueError(f"Service {service} not registered")

        async with httpx.AsyncClient() as client:
            response = await client.post(
                f"{url}{endpoint}",
                json=data,
                headers={'Content-Type': 'application/json'}
            )
            return response.json()

    async def orchestrate_workflow(self, workflow: list) -> Dict:
        results = {}
        current_data = {}

        for step in workflow:
            output = await self.call_service(
                step['service'],
                step['endpoint'],
                step.get('input', current_data)
            )
            results[step.get('name', step['service'])] = output
            current_data = output

        return results
